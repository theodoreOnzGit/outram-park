\section{The Vision}
% =============================================================================

\begin{frame}{OUTRAM PARK in one sentence}
    \begin{center}
        \large
        A \textbf{one-stop, open-source multiphysics toolkit} where researchers
        collaborate on \textbf{anything nuclear}, on one common, unified
        codebase that is \textbf{drafted with AI and verified by humans}.
    \end{center}

    \medskip
    \begin{itemize}
        \item \textbf{O}pen-source \textbf{U}nified \textbf{TRA}nsient
              \textbf{M}ulti-Physics \textbf{A}dvanced \textbf{R}eactor
              simulation \textbf{K}it.
        \item Neutronics, thermal hydraulics and CFD, fuel and structures,
              materials, chemistry and source term in \textbf{one Rust
              workspace}, with one API style and one V\&V discipline.
        \item \textbf{Long-run claim:} OUTRAM PARK can model \emph{any}
              Gen IV reactor.
        \item \textbf{Endgame:} a self-sustaining community, as OpenFOAM
              has.
    \end{itemize}
\end{frame}

\begin{frame}{The problem it addresses}
    Nuclear multiphysics today is \textbf{fragmented}:

    \medskip
    \begin{itemize}
        \item Each phenomenon lives in a different code, with its own
              language, input format, licence and access regime.
        \item Coupling them is a research project in itself, and is often
              redone by every group.
        \item Many reference codes are export-controlled or closed, which is
              hard on students, small groups and new nuclear countries.
        \item Building input decks for system codes such as MELCOR or ASTEC is slow, so few
              scenarios are ever explored.
    \end{itemize}

    \medskip
    \textbf{OUTRAM PARK's answer:} port the established open codes into one
    workspace, measure each port against its upstream, and put everything
    behind a common API that humans (and AI agents) can actually use.
\end{frame}

\begin{frame}{Why now: AI drafts, humans verify}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \textbf{AI makes the breadth possible}
            \begin{itemize}
                \footnotesize
                \item Translating OpenFOAM, OpenMC, NJOY2016, CoolProp,
                      OFFBEAT, \dots\ into one language was never
                      affordable for one group.
                \item AI drafting has produced $\sim$1.27M lines across the
                      workspace in under two years.
            \end{itemize}
        \end{column}
        \begin{column}{0.48\textwidth}
            \textbf{Humans make it trustworthy}
            \begin{itemize}
                \footnotesize
                \item Everything is \textbf{unverified until validated}.
                \item AI-generated V\&V is a \emph{screening gate}, not a
                      verdict.
                \item A code is \textbf{ready} only after humans have studied
                      it and reproduced its cases themselves.
            \end{itemize}
        \end{column}
    \end{columns}

    \medskip
    \footnotesize
    The two halves need each other. AI without human V\&V is untrustworthy;
    human effort without AI drafting is too slow to cover a whole reactor.
\end{frame}

\section{Who It Serves}
% =============================================================================

\begin{frame}{Four groups, one codebase}
    \roadtab{
    \begin{tabular}{L{0.2\textwidth}L{0.46\textwidth}L{0.24\textwidth}}
        \toprule
        \textbf{Group} & \textbf{What they do with it} & \textbf{Priority} \\
        \midrule
        \textbf{Researchers} & High-fidelity study of phenomena: neutronics,
            boiling, precursor drift, double heterogeneity, Kinchin--Pease
            damage, stress corrosion cracking
            & \textbf{Now} \\[2pt]
        \textbf{Digital twins} & Real-time simulators on cheap physics models
            and ML/ROM surrogates; optionally coupled to integral effects tests
            & \textbf{Now} \\[2pt]
        \textbf{Students \& public} & Outreach simulators with accident
            scenarios; undergrad and postgrad projects and theses in the
            source code
            & \textbf{Now} \\[2pt]
        \textbf{Safety analysts} & Cheap accident-scenario exploration ahead of
            MELCOR / ASTEC
            & \textbf{When SEMBAWANG is reviewed} \\
        \bottomrule
    \end{tabular}}
\end{frame}

\begin{frame}{How the four groups feed each other}
    \begin{center}
        \footnotesize
        \setlength{\tabcolsep}{6pt}
        \begin{tabular}{ccccc}
            \textbf{High-fidelity codes} & $\longrightarrow$ &
            \textbf{Surrogates} & $\longrightarrow$ &
            \textbf{Twins \& simulators} \\
            {\scriptsize neutronics, CFD, fuel} & &
            {\scriptsize cheap physics + ML/ROM} & &
            {\scriptsize real time, MQTT / OPC-UA} \\[6pt]
            $\big\uparrow$ & & & & $\big\downarrow$ \\[2pt]
            \textbf{Students \& researchers} & & $\longleftarrow$ & &
            \textbf{Outreach \& teaching} \\
            {\scriptsize projects, theses, human V\&V} & & & &
            {\scriptsize accident scenarios}
        \end{tabular}
    \end{center}

    \medskip
    \begin{itemize}
        \item High-fidelity runs produce the \textbf{training data} for
              ML/ROM surrogates.
        \item Surrogates and cheap models power the \textbf{twins} and the
              \textbf{outreach simulators}.
        \item Outreach draws in \textbf{students}, whose projects on the
              source code become \textbf{human V\&V}, which makes the
              high-fidelity codes ready.
    \end{itemize}
\end{frame}

\begin{frame}{Researchers and students: into the source}
    \begin{itemize}
        \item \textbf{Researchers} study phenomena at high fidelity, on the
              same codebase as their collaborators, rather than coupling five
              codes by hand.
        \item \textbf{Undergraduates and postgraduates} work directly in the
              source code through final-year projects, course projects and
              theses.
        \item Their hand-written cases \emph{are} human V\&V: every student
              project that reproduces a benchmark through the API moves a
              crate toward ready.
        \item API friction they hit is recorded and fixed. The API is
              designed to be usable by a newcomer (and by a small AI model)
              without reading the source.
    \end{itemize}

    \medskip
    \footnotesize
    For \textbf{public outreach}, prepared simulators with scripted accident
    scenarios fit better than source code. These build on the teaching decks
    already in \texttt{presentations/}.
\end{frame}

\begin{frame}{Digital twins: two kinds of surrogate}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \textbf{Cheap physics models}
            \begin{itemize}
                \footnotesize
                \item Point kinetics (\texttt{teh-o-prke}), 1-D thermal
                      hydraulics (\texttt{tuas}), steam cycles
                      (\texttt{tampines}).
                \item Run in real time today: the FHR, HTGR and CIET
                      simulators.
            \end{itemize}
        \end{column}
        \begin{column}{0.48\textwidth}
            \textbf{ML / ROM surrogates}
            \begin{itemize}
                \footnotesize
                \item Trained on OUTRAM PARK's own high-fidelity codes.
                \item Bring higher-fidelity physics to real time.
            \end{itemize}
        \end{column}
    \end{columns}

    \medskip
    \textbf{Deployment path:}
    \begin{enumerate}
        \item The twin is a \textbf{real-time simulator} first.
        \item If desired, it \textbf{connects to an integral effects test}
              facility over MQTT and OPC-UA. CIET v2 already ships an
              OPC-UA server.
    \end{enumerate}

    \medskip
    \footnotesize
    Offline research and education only. Never connected to operating plant
    or safety systems.
\end{frame}

\begin{frame}{Safety analysis: SEMBAWANG ahead of MELCOR / ASTEC}
    \textbf{SEMBAWANG} (\textbf{S}evere-accident \textbf{E}volution and
    \textbf{M}elt \textbf{B}ehaviour \textbf{A}nalysis \textbf{W}orkbench for
    \textbf{A}dvanced \textbf{N}uclear \textbf{G}eometries) is the
    \emph{cheap} model.

    \medskip
    \roadtab{
    \begin{tabular}{cL{0.78\textwidth}}
        \toprule
        1 & AI + SEMBAWANG \textbf{explore many accident scenarios} quickly. \\
        2 & \textbf{Narrow down} to the scenarios that actually need a
            full assessment. \\
        3 & \textbf{Verify the parameters} in SEMBAWANG models, which are
            quick to build. \\
        4 & \textbf{Compile the information} the MELCOR / ASTEC input decks
            need. \\
        5 & Engineers build the decks; \textbf{real assessment begins} in
            MELCOR / ASTEC. \\
        \bottomrule
    \end{tabular}}

    \medskip
    \footnotesize
    \textbf{Status:} TRISO source term implemented (via \texttt{boon-lay});
    accident progression not yet. It orchestrates the offsite chain:
    \texttt{changi} (dispersion) and \texttt{redhill} (ground transport). No
    dose is computed. Under AI drafting, needs review; this group waits until
    it is ready.
\end{frame}

\section{The V\&V Ladder}
% =============================================================================

\begin{frame}{When is a code ``ready''?}
    \roadtab{
    \begin{tabular}{cL{0.3\textwidth}L{0.52\textwidth}}
        \toprule
        \textbf{Step} & \textbf{Gate} & \textbf{What happens} \\
        \midrule
        1 & AI: code-to-code & Verified against its upstream \emph{during}
            translation (e.g.\ NJOY2016 to 7 significant figures). \\[2pt]
        2 & AI: test reactor & One class of test reactor per category,
            reproduced by an AI-generated V\&V case. \\[2pt]
        3 & Human V\&V & Humans study the code and API, then
            \textbf{hand-code the same cases} through the user-friendly Rust
            API. Friction leads to API upgrades. \\[2pt]
        4 & \textbf{Ready} & The maintainer signs off \textbf{both axes}
            in the crate's README: V\&V, and human / user interface. \\
        \bottomrule
    \end{tabular}}

    \medskip
    \footnotesize
    Passing steps 1--2 is \textbf{proof of concept}: the code is functional
    to a large extent and ready for human eyes. It is not a verdict.
\end{frame}

\begin{frame}{What each step produces}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \textbf{AI drafts (steps 1--2)}
            \begin{itemize}
                \footnotesize
                \item Proof that the code runs the reactor class.
                \item \textbf{Publishable:} initial V\&V papers, and
                      software-engineering methodology papers on AI-drafted,
                      human-verified scientific code.
                \item The signal to start human review.
            \end{itemize}
        \end{column}
        \begin{column}{0.48\textwidth}
            \textbf{Human V\&V (step 3)}
            \begin{itemize}
                \footnotesize
                \item Independent reproduction of the cases.
                \item \textbf{An API tested by use:} every friction point is
                      a recorded fix.
                \item \textbf{Publishable:} the definitive V\&V papers.
                \item Natural student projects and theses.
            \end{itemize}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{First wave: four families, neutronics + TH/CFD}
    \roadtab{
    \begin{tabular}{lL{0.28\textwidth}L{0.36\textwidth}}
        \toprule
        \textbf{Family} & \textbf{Test reactor / facility} & \textbf{Notes} \\
        \midrule
        Pebble bed      & \textbf{HTR-10}, \textbf{HTTR} & Double heterogeneity;
                          HTR-10 gate exists in \texttt{outram-mc-libs} \\
        Molten salt     & \textbf{MSRE}   & Precursor drift (Moltres fork) \\
        PWR (integral)  & \textbf{MASLWR} (Oregon State) & TH only: electrically
                          heated; IAEA natural-circulation benchmark \\
        PWR neutronics  & \emph{open} & e.g.\ BEAVRS \\
        BWR             & \emph{open} & e.g.\ Peach Bottom turbine trip; needs
                          boiling / two-phase \\
        \bottomrule
    \end{tabular}}

    \medskip
    \footnotesize
    Each family is taken up the full V\&V ladder, for \textbf{both}
    neutronics and thermal hydraulics/CFD.
\end{frame}

\begin{frame}{Later waves: toward any Gen IV reactor}
    \begin{itemize}
        \item \textbf{Next: fast / liquid-metal.} EBR-I as the test reactor,
              pursued mainly through \textbf{research collaborations}.
        \item \textbf{Then the rest of Gen IV:} gas-cooled fast,
              lead-cooled, supercritical water, as collaborators and
              funding allow.
        \item Each wave reuses the ladder: code-to-code, one test reactor,
              human V\&V, ready.
    \end{itemize}

    \medskip
    \textbf{The claim to earn:} \emph{OUTRAM PARK can model any Gen IV
    reactor}, with a human-verified test case behind each family.
\end{frame}

\section{Community}
% =============================================================================

\begin{frame}{A community to sustain it}
    \textbf{Goal:} OUTRAM PARK outlives any one maintainer, as OpenFOAM
    does, sustained by an open community across:

    \medskip
    \begin{center}
        \textbf{Nuclear} \quad$\cdot$\quad \textbf{Chemical} \quad$\cdot$\quad
        \textbf{Mechanical} \quad$\cdot$\quad \textbf{Materials}
        \\ science and engineering
    \end{center}

    \medskip
    \begin{itemize}
        \item \textbf{Needed in the next year or two:} more people, above
              all human V\&V reviewers.
        \item \textbf{Already in place:} GPL-3.0, \texttt{GOVERNANCE.md},
              \texttt{CONTRIBUTING.md}, responsible-use and provenance
              policies.
        \item \textbf{To build:} contributor onboarding, multi-institution
              maintainership, a user base beyond SNRSI.
    \end{itemize}
\end{frame}

\begin{frame}{How to get involved}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \textbf{Collaborators \& researchers}
            \begin{itemize}
                \footnotesize
                \item Take a test reactor through human V\&V and co-author
                      the paper.
                \item Bring a phenomenon or a Gen IV family (e.g.\ EBR-I).
                \item Supervise student projects on the codebase.
            \end{itemize}

            \medskip
            \textbf{Contributors}
            \begin{itemize}
                \footnotesize
                \item Review AI drafts; report API friction; write cases.
            \end{itemize}
        \end{column}
        \begin{column}{0.48\textwidth}
            \textbf{Funders}
            \begin{itemize}
                \footnotesize
                \item Human V\&V effort: people, not compute, are the
                      bottleneck.
                \item Students and postdocs on first-wave families.
                \item Collaboration on the next reactor wave.
            \end{itemize}

            \medskip
            \textbf{Start here}
            \begin{itemize}
                \footnotesize
                \item \texttt{github.com/theodoreOnzGit/outram-park}
                \item \texttt{github.com/theodoreOnzGit/outram-park-backend}
            \end{itemize}
        \end{column}
    \end{columns}
\end{frame}
